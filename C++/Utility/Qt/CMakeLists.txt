cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(CameraPresentation LANGUAGES CXX)
endif()
if(TARGET CameraPresentation)
    return()
endif()
if(TARGET Qt6::Widgets)
    set(presentation_qt Qt6)
elseif(TARGET Qt5::Widgets)
    set(presentation_qt Qt5)
else()
    find_package(Qt6 REQUIRED COMPONENTS Widgets)
    set(presentation_qt Qt6)
endif()
add_library(CameraPresentation STATIC CameraControlUi.cpp CameraControlUi.h)
set_target_properties(CameraPresentation PROPERTIES AUTOMOC ON POSITION_INDEPENDENT_CODE ON)
target_compile_features(CameraPresentation PUBLIC cxx_std_17)
target_compile_options(CameraPresentation PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
target_include_directories(CameraPresentation PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(CameraPresentation PUBLIC ${presentation_qt}::Widgets)
